# Audio transcription (L3) — Basic-Pitch (pulls TensorFlow ~1GB).
# Enables mp3/wav input. F1≈0.43: good for clean single-instrument audio,
# rough for dense mixes (pair with demucs below for full songs).
basic_pitch

# Source separation (L2) — Demucs (pulls PyTorch ~800MB + model weights).
# torchaudio must be present too (demucs imports it); torchaudio 2.9+ moved
# audio decoding to torchcodec, so separate.py reads files via soundfile
# instead and only needs torchaudio for resampling. Install with uv:
# Splits a full-song mix into vocals/bass/other stems before transcription,
# materially improving mp3 quality. Optional but recommended for full songs.
demucs

# Beat tracking (L4) — used by experiments/bench_beats.py. Installed, but NOT
# yet wired into analysis.py: on the only audio available here (our own
# synthesized render) it got the tempo right and the beat phase wrong, which is
# not a usable verdict. See README §8 P0 for why this is blocked on real songs.
beat-this

# Optional (NOT installed by default):
#   MuScriptor (2026 AMT, Multi-F1 47.8 vs YourMT3+ 21.9; weights are
#   CC BY-NC 4.0, so it cannot ship in a commercial product without a swap):
#     pip install muscriptor
#   RMVPE ONNX (monophonic f0 for the vocal/bass stems — near-perfect where
#   polyphonic AMT collapses; Apache-2.0, small, fast on Apple Silicon):
#     pip install rmvpe-onnx
#   MSST (BS-RoFormer separation; demucs' upstream is archived since 2023):
